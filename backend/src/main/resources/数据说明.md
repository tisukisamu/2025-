## 📝 数据验证

导入完成后，可以运行以下SQL验证数据：

```sql
-- 查看各表数据量
SELECT 
    (SELECT COUNT(*) FROM users) AS users_count,
    (SELECT COUNT(*) FROM teachers) AS teachers_count,
    (SELECT COUNT(*) FROM students) AS students_count,
    (SELECT COUNT(*) FROM courses) AS courses_count,
    (SELECT COUNT(*) FROM schedules) AS schedules_count,
    (SELECT COUNT(*) FROM enrollments) AS enrollments_count,
    (SELECT COUNT(*) FROM payments) AS payments_count,
    (SELECT COUNT(*) FROM bills) AS bills_count,
    (SELECT COUNT(*) FROM notifications) AS notifications_count,
    (SELECT COUNT(*) FROM settings) AS settings_count;

-- 查看课程报名情况
SELECT 
    c.name AS 课程名称,
    t.name AS 授课教师,
    c.price AS 价格,
    c.capacity AS 容量,
    c.enrolled_count AS 已报名人数
FROM courses c
LEFT JOIN teachers t ON c.teacher_id = t.id
ORDER BY c.id;

-- 查看学员报名情况
SELECT 
    s.name AS 学员姓名,
    c.name AS 课程名称,
    e.status AS 报名状态,
    e.payment_status AS 支付状态
FROM enrollments e
LEFT JOIN students s ON e.student_id = s.id
LEFT JOIN courses c ON e.course_id = c.id
ORDER BY e.id;
```
